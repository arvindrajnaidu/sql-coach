Cheat sheet

-- Order of clauses
SELECT   columns / aggregates
FROM     table
JOIN     other ON key = key
WHERE    row filter
GROUP BY columns
HAVING   group filter
ORDER BY columns [DESC]
LIMIT    n;
-- Filters
= != < > <= >=
AND  OR  NOT
IN ('a','b')
BETWEEN x AND y
LIKE 'A%'   -- % any, _ one char
IS NULL / IS NOT NULL
-- Aggregates
COUNT(*)  COUNT(col)  COUNT(DISTINCT col)
SUM  AVG  MIN  MAX
ROUND(x, 2)
100.0 * a / b   -- avoid integer division
-- Joins
JOIN      only matches
LEFT JOIN keep all left rows
          (right side NULL if no match)
-- anti-join: LEFT JOIN ... WHERE right.id IS NULL
-- Conditional logic
CASE WHEN x THEN 'a'
     WHEN y THEN 'b'
     ELSE 'c' END
SUM(CASE WHEN cond THEN 1 ELSE 0 END)
-- CTE and windows
WITH t AS (SELECT ...)
SELECT * FROM t;

ROW_NUMBER() OVER (PARTITION BY g ORDER BY x DESC)
RANK() / DENSE_RANK()
SUM(x) OVER (ORDER BY d)  -- running total
-- Dates (SQLite)
strftime('%Y-%m', d)   -- '2024-03'
date(d, '+7 days')
-- Postgres: DATE_TRUNC('month', d)
